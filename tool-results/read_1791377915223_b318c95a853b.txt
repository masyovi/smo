   446→function ActionPanel({
   447→  report,
   448→  currentUserId,
   449→  canAssign,
   450→  onUpdated,
   451→}: {
   452→  report: ReportDetail
   453→  currentUserId: string
   454→  canAssign: boolean
   455→  onUpdated: () => void
   456→}) {
   457→  const qc = useQueryClient()
   458→  const [status, setStatus] = React.useState<ReportStatus>(report.status)
   459→  const [priority, setPriority] = React.useState<ReportPriority>(report.priority)
   460→  const [resolution, setResolution] = React.useState(report.resolution ?? '')
   461→  const [assigneeId, setAssigneeId] = React.useState<string>(report.assigneeId ?? '__none__')
   462→  const [submitting, setSubmitting] = React.useState(false)
   463→
   464→  // Keep local state synced when report changes
   465→  React.useEffect(() => {
   466→    setStatus(report.status)
   467→    setPriority(report.priority)
   468→    setResolution(report.resolution ?? '')
   469→    setAssigneeId(report.assigneeId ?? '__none__')
   470→  }, [report.id, report.status, report.priority, report.resolution, report.assigneeId])
   471→
   472→  const { data: technicians } = useQuery<{
   473→    id: string
   474→    name: string
   475→    email: string
   476→    role: string
   477→  }[]>({
   478→    queryKey: ['technicians'],
   479→    queryFn: async () => {
   480→      const all = await apiFetch<Array<{ id: string; name: string; email: string; role: string }>>(
   481→        '/api/users'
   482→      )
   483→      return all.filter((u) => u.role === 'TECHNICIAN' || u.role === 'ADMIN')
   484→    },
   485→    // Only fetch the user list when the caller actually has permission to assign.
   486→    enabled: canAssign,
   487→  })
   488→
   489→  const statusChanged = status !== report.status
   490→  const priorityChanged = priority !== report.priority
   491→  const resolutionChanged = resolution !== (report.resolution ?? '')
   492→  const assigneeChanged =
   493→    (assigneeId === '__none__' ? null : assigneeId) !==
   494→    (report.assigneeId ?? null)
   495→
   496→  const hasChange =
   497→    statusChanged || priorityChanged || resolutionChanged || assigneeChanged
   498→
   499→  async function handleSave() {
   500→    setSubmitting(true)
   501→    const body: Record<string, unknown> = {}
   502→    if (statusChanged) body.status = status
   503→    if (priorityChanged) body.priority = priority
   504→    if (resolutionChanged) body.resolution = resolution
   505→    if (assigneeChanged) body.assigneeId = assigneeId === '__none__' ? null : assigneeId
   506→    try {
   507→      await apiFetch(`/api/reports/${report.id}`, {
   508→        method: 'PATCH',
   509→        body: JSON.stringify(body),
   510→      })
   511→      toast.success('Perubahan disimpan')
   512→      onUpdated()
   513→    } catch (e) {
   514→      const msg = e instanceof ApiError ? e.message : 'Gagal menyimpan'
   515→      toast.error('Gagal menyimpan', { description: msg })
   516→    } finally {
   517→      setSubmitting(false)
   518→    }
   519→  }
   520→
   521→  return (
   522→    <Card>
   523→      <CardHeader className="pb-3">
   524→        <CardTitle className="flex items-center gap-2 text-base">
   525→          <Wrench className="size-4" />
   526→          Panel Aksi
   527→        </CardTitle>
   528→        <CardDescription className="text-xs">
   529→          Perbarui status, prioritas, resolusi, atau penugasan laporan
   530→        </CardDescription>
   531→      </CardHeader>
   532→      <CardContent className="grid gap-4 px-4 sm:px-6 md:grid-cols-2">
   533→        <div className="space-y-2">
   534→          <Label htmlFor="status">Status</Label>
   535→          <Select value={status} onValueChange={(v) => setStatus(v as ReportStatus)}>
   536→            <SelectTrigger id="status" className="w-full">
   537→              <SelectValue />
   538→            </SelectTrigger>
   539→            <SelectContent>
   540→              {STATUS_LIST.map((s) => (
   541→                <SelectItem key={s} value={s}>
   542→                  <span className="inline-flex items-center gap-2">
   543→                    <span
   544→                      className={cn('size-1.5 rounded-full', STATUS_CONFIG[s].dot)}
   545→                    />
   546→                    {STATUS_CONFIG[s].label}
   547→                  </span>
   548→                </SelectItem>
   549→              ))}
   550→            </SelectContent>
   551→          </Select>
   552→        </div>
   553→
   554→        <div className="space-y-2">
   555→          <Label htmlFor="priority">Prioritas</Label>
   556→          <Select value={priority} onValueChange={(v) => setPriority(v as ReportPriority)}>
   557→            <SelectTrigger id="priority" className="w-full">
   558→              <SelectValue />
   559→            </SelectTrigger>
   560→            <SelectContent>
   561→              {PRIORITY_LIST.map((p) => (
   562→                <SelectItem key={p} value={p}>
   563→                  {PRIORITY_CONFIG[p].label}
   564→                </SelectItem>
   565→              ))}
   566→            </SelectContent>
   567→          </Select>
   568→        </div>
   569→
   570→        {canAssign && (
   571→          <div className="space-y-2 md:col-span-2">
   572→            <Label htmlFor="assignee">Penugasan Teknisi</Label>
   573→            <Select value={assigneeId} onValueChange={(v) => setAssigneeId(v)}>
   574→              <SelectTrigger id="assignee" className="w-full">
   575→                <SelectValue placeholder="Pilih teknisi" />
   576→              </SelectTrigger>
   577→              <SelectContent>
   578→                <SelectItem value="__none__">Tidak ditugaskan</SelectItem>
   579→                {technicians?.map((t) => (
   580→                  <SelectItem key={t.id} value={t.id}>
   581→                    {t.name} ({ROLE_CONFIG[t.role as UserRole].label})
   582→                  </SelectItem>
   583→                ))}
   584→              </SelectContent>
   585→            </Select>
   586→          </div>
   587→        )}
   588→
   589→        <div className="space-y-2 md:col-span-2">
   590→          <Label htmlFor="resolution">Resolusi</Label>
   591→          <Textarea
   592→            id="resolution"
   593→            value={resolution}
   594→            onChange={(e) => setResolution(e.target.value)}
   595→            placeholder="Jelaskan langkah penyelesaian, temuan, atau catatan teknis…"
   596→            rows={3}
   597→          />
   598→        </div>
   599→
   600→        <div className="md:col-span-2 flex items-center justify-between gap-2">
   601→          <span className="text-muted-foreground text-xs">
   602→            {hasChange ? 'Ada perubahan belum disimpan' : 'Tidak ada perubahan'}

   614→function OwnerEditPanel({
   615→  report,
   616→  onUpdated,
   617→}: {
   618→  report: ReportDetail
   619→  onUpdated: () => void
   620→}) {
   621→  const [open, setOpen] = React.useState(false)
   622→  const [title, setTitle] = React.useState(report.title)
   623→  const [description, setDescription] = React.useState(report.description)
   624→  const [submitting, setSubmitting] = React.useState(false)
   625→
   626→  React.useEffect(() => {
   627→    setTitle(report.title)
   628→    setDescription(report.description)
   629→  }, [report.id, report.title, report.description])
   630→
   631→  const changed = title !== report.title || description !== report.description
   632→
   633→  async function handleSave() {
   634→    setSubmitting(true)
   635→    try {
   636→      await apiFetch(`/api/reports/${report.id}`, {
   637→        method: 'PATCH',
   638→        body: JSON.stringify({ title, description }),
   639→      })
   640→      toast.success('Laporan diperbarui')
   641→      onUpdated()
   642→      setOpen(false)
   643→    } catch (e) {
   644→      const msg = e instanceof ApiError ? e.message : 'Gagal menyimpan'
   645→      toast.error('Gagal menyimpan', { description: msg })
   646→    } finally {
   647→      setSubmitting(false)
   648→    }
   649→  }
   650→
   651→  return (
   652→    <Dialog open={open} onOpenChange={setOpen}>
   653→      <DialogTrigger asChild>
   654→        <Button variant="outline" className="w-full justify-start">
   655→          <Pencil className="size-4" />
   656→          Edit laporan saya
   657→        </Button>
   658→      </DialogTrigger>
   659→      <DialogContent>
   660→        <DialogHeader>
   661→          <DialogTitle>Edit Laporan</DialogTitle>
   662→          <DialogDescription>
   663→            Anda dapat mengedit judul dan deskripsi selama laporan masih berstatus PENDING.
   664→          </DialogDescription>
   665→        </DialogHeader>
   666→        <div className="space-y-3">
   667→          <div className="space-y-2">
   668→            <Label htmlFor="edit-title">Judul</Label>
   669→            <Input id="edit-title" value={title} onChange={(e) => setTitle(e.target.value)} />
   670→          </div>
   671→          <div className="space-y-2">
   672→            <Label htmlFor="edit-desc">Deskripsi</Label>
   673→            <Textarea
   674→              id="edit-desc"
   675→              rows={5}
   676→              value={description}
   677→              onChange={(e) => setDescription(e.target.value)}
   678→            />
   679→          </div>
   680→        </div>
   681→        <DialogFooter>
   682→          <Button variant="outline" onClick={() => setOpen(false)}>
   683→            Batal
   684→          </Button>
   685→          <Button onClick={handleSave} disabled={!changed || submitting}>
   686→            {submitting && <Loader2 className="size-4 animate-spin" />}
   687→            Simpan
   688→          </Button>
   689→        </DialogFooter>
   690→      </DialogContent>
   691→    </Dialog>
   692→  )
   693→}

   695→function CommentComposer({
   696→  reportId,
   697→  onCommented,
   698→}: {
   699→  reportId: string
   700→  onCommented: () => void
   701→}) {
   702→  const [message, setMessage] = React.useState('')
   703→  const [submitting, setSubmitting] = React.useState(false)
   704→
   705→  async function handleSend() {
   706→    const text = message.trim()
   707→    if (!text) return
   708→    setSubmitting(true)
   709→    try {
   710→      await apiFetch(`/api/reports/${reportId}/comments`, {
   711→        method: 'POST',
   712→        body: JSON.stringify({ message: text }),
   713→      })
   714→      setMessage('')
   715→      toast.success('Komentar ditambahkan')
   716→      onCommented()
   717→    } catch (e) {
   718→      const msg = e instanceof ApiError ? e.message : 'Gagal menambah komentar'
   719→      toast.error('Gagal menambah komentar', { description: msg })
   720→    } finally {
   721→      setSubmitting(false)
   722→    }
   723→  }
   724→
   725→  return (
   726→    <Card>
   727→      <CardHeader className="pb-3">
   728→        <CardTitle className="flex items-center gap-2 text-base">
   729→          <MessageSquare className="size-4" />
   730→          Tambah Komentar
   731→        </CardTitle>
   732→      </CardHeader>
   733→      <CardContent className="space-y-3 px-4 sm:px-6">
   734→        <Textarea
   735→          value={message}
   736→          onChange={(e) => setMessage(e.target.value)}
   737→          placeholder="Tulis komentar atau update singkat untuk laporan ini…"
   738→          rows={3}
   739→          onKeyDown={(e) => {
   740→            if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   741→              handleSend()
   742→            }
   743→          }}
   744→        />
   745→        <div className="flex items-center justify-between gap-2">
   746→          <span className="text-muted-foreground text-[11px]">
   747→            Tekan Cmd/Ctrl + Enter untuk kirim cepat
   748→          </span>
   749→          <Button onClick={handleSend} disabled={!message.trim() || submitting}>
   750→            {submitting ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   751→            Kirim Komentar
   752→          </Button>
   753→        </div>
   754→      </CardContent>
   755→    </Card>
   756→  )
   757→}
