   426→function ActionPanel({
   427→  report,
   428→  currentUserId,
   429→  onUpdated,
   430→}: {
   431→  report: ReportDetail
   432→  currentUserId: string
   433→  onUpdated: () => void
   434→}) {
   435→  const qc = useQueryClient()
   436→  const [status, setStatus] = React.useState<ReportStatus>(report.status)
   437→  const [priority, setPriority] = React.useState<ReportPriority>(report.priority)
   438→  const [resolution, setResolution] = React.useState(report.resolution ?? '')
   439→  const [assigneeId, setAssigneeId] = React.useState<string>(report.assigneeId ?? '__none__')
   440→  const [submitting, setSubmitting] = React.useState(false)
   441→
   442→  // Keep local state synced when report changes
   443→  React.useEffect(() => {
   444→    setStatus(report.status)
   445→    setPriority(report.priority)
   446→    setResolution(report.resolution ?? '')
   447→    setAssigneeId(report.assigneeId ?? '__none__')
   448→  }, [report.id, report.status, report.priority, report.resolution, report.assigneeId])
   449→
   450→  const { data: technicians } = useQuery<{
   451→    id: string
   452→    name: string
   453→    email: string
   454→    role: string
   455→  }[]>({
   456→    queryKey: ['technicians'],
   457→    queryFn: async () => {
   458→      const all = await apiFetch<Array<{ id: string; name: string; email: string; role: string }>>(
   459→        '/api/users'
   460→      )
   461→      return all.filter((u) => u.role === 'TECHNICIAN' || u.role === 'ADMIN')
   462→    },
   463→    enabled: useAppStore.getState().user?.role === 'ADMIN',
   464→  })
   465→
   466→  const statusChanged = status !== report.status
   467→  const priorityChanged = priority !== report.priority
   468→  const resolutionChanged = resolution !== (report.resolution ?? '')
   469→  const assigneeChanged =
   470→    (assigneeId === '__none__' ? null : assigneeId) !==
   471→    (report.assigneeId ?? null)
   472→
   473→  const hasChange =
   474→    statusChanged || priorityChanged || resolutionChanged || assigneeChanged
   475→
   476→  async function handleSave() {
   477→    setSubmitting(true)
   478→    const body: Record<string, unknown> = {}
   479→    if (statusChanged) body.status = status
   480→    if (priorityChanged) body.priority = priority
   481→    if (resolutionChanged) body.resolution = resolution
   482→    if (assigneeChanged) body.assigneeId = assigneeId === '__none__' ? null : assigneeId
   483→    try {
   484→      await apiFetch(`/api/reports/${report.id}`, {
   485→        method: 'PATCH',
   486→        body: JSON.stringify(body),
   487→      })
   488→      toast.success('Perubahan disimpan')
   489→      onUpdated()
   490→    } catch (e) {
   491→      const msg = e instanceof ApiError ? e.message : 'Gagal menyimpan'
   492→      toast.error('Gagal menyimpan', { description: msg })
   493→    } finally {
   494→      setSubmitting(false)
   495→    }
   496→  }
   497→
   498→  return (
   499→    <Card>
   500→      <CardHeader className="pb-3">
   501→        <CardTitle className="flex items-center gap-2 text-base">
   502→          <Wrench className="size-4" />
   503→          Panel Aksi
   504→        </CardTitle>
   505→        <CardDescription className="text-xs">
   506→          Perbarui status, prioritas, resolusi, atau penugasan laporan
   507→        </CardDescription>
   508→      </CardHeader>
   509→      <CardContent className="grid gap-4 px-4 sm:px-6 md:grid-cols-2">
   510→        <div className="space-y-2">
   511→          <Label htmlFor="status">Status</Label>
   512→          <Select value={status} onValueChange={(v) => setStatus(v as ReportStatus)}>
   513→            <SelectTrigger id="status" className="w-full">
   514→              <SelectValue />
   515→            </SelectTrigger>
   516→            <SelectContent>
   517→              {STATUS_LIST.map((s) => (
   518→                <SelectItem key={s} value={s}>
   519→                  <span className="inline-flex items-center gap-2">
   520→                    <span
   521→                      className={cn('size-1.5 rounded-full', STATUS_CONFIG[s].dot)}
   522→                    />
   523→                    {STATUS_CONFIG[s].label}
   524→                  </span>
   525→                </SelectItem>
   526→              ))}
   527→            </SelectContent>
   528→          </Select>
   529→        </div>
   530→
   531→        <div className="space-y-2">
   532→          <Label htmlFor="priority">Prioritas</Label>
   533→          <Select value={priority} onValueChange={(v) => setPriority(v as ReportPriority)}>
   534→            <SelectTrigger id="priority" className="w-full">
   535→              <SelectValue />
   536→            </SelectTrigger>
   537→            <SelectContent>
   538→              {PRIORITY_LIST.map((p) => (
   539→                <SelectItem key={p} value={p}>
   540→                  {PRIORITY_CONFIG[p].label}
   541→                </SelectItem>
   542→              ))}
   543→            </SelectContent>
   544→          </Select>
   545→        </div>
   546→
   547→        {useAppStore.getState().user?.role === 'ADMIN' && (
   548→          <div className="space-y-2 md:col-span-2">
   549→            <Label htmlFor="assignee">Penugasan Teknisi</Label>
   550→            <Select value={assigneeId} onValueChange={(v) => setAssigneeId(v)}>
   551→              <SelectTrigger id="assignee" className="w-full">
   552→                <SelectValue placeholder="Pilih teknisi" />
   553→              </SelectTrigger>
   554→              <SelectContent>
   555→                <SelectItem value="__none__">Tidak ditugaskan</SelectItem>
   556→                {technicians?.map((t) => (
   557→                  <SelectItem key={t.id} value={t.id}>
   558→                    {t.name} ({ROLE_CONFIG[t.role as UserRole].label})
   559→                  </SelectItem>
   560→                ))}
   561→              </SelectContent>
   562→            </Select>
   563→          </div>
   564→        )}
   565→
   566→        <div className="space-y-2 md:col-span-2">
   567→          <Label htmlFor="resolution">Resolusi</Label>
   568→          <Textarea
   569→            id="resolution"
   570→            value={resolution}
   571→            onChange={(e) => setResolution(e.target.value)}
   572→            placeholder="Jelaskan langkah penyelesaian, temuan, atau catatan teknis…"
   573→            rows={3}
   574→          />
   575→        </div>
   576→
   577→        <div className="md:col-span-2 flex items-center justify-between gap-2">
   578→          <span className="text-muted-foreground text-xs">
   579→            {hasChange ? 'Ada perubahan belum disimpan' : 'Tidak ada perubahan'}
   580→          </span>
   581→          <Button onClick={handleSave} disabled={!hasChange || submitting}>
   582→            {submitting && <Loader2 className="size-4 animate-spin" />}
   583→            Simpan Perubahan
   584→          </Button>
   585→        </div>
   586→      </CardContent>
   587→    </Card>
   588→  )
   589→}

   600→  const [description, setDescription] = React.useState(report.description)
   601→  const [submitting, setSubmitting] = React.useState(false)
   602→
   603→  React.useEffect(() => {
   604→    setTitle(report.title)
   605→    setDescription(report.description)
   606→  }, [report.id, report.title, report.description])
   607→
   608→  const changed = title !== report.title || description !== report.description
   609→
   610→  async function handleSave() {
   611→    setSubmitting(true)
   612→    try {
   613→      await apiFetch(`/api/reports/${report.id}`, {
   614→        method: 'PATCH',
   615→        body: JSON.stringify({ title, description }),
   616→      })
   617→      toast.success('Laporan diperbarui')
   618→      onUpdated()
   619→      setOpen(false)
   620→    } catch (e) {
   621→      const msg = e instanceof ApiError ? e.message : 'Gagal menyimpan'
   622→      toast.error('Gagal menyimpan', { description: msg })
   623→    } finally {
   624→      setSubmitting(false)
   625→    }
   626→  }
   627→
   628→  return (
   629→    <Dialog open={open} onOpenChange={setOpen}>
   630→      <DialogTrigger asChild>
   631→        <Button variant="outline" className="w-full justify-start">
   632→          <Pencil className="size-4" />
   633→          Edit laporan saya
   634→        </Button>
   635→      </DialogTrigger>
   636→      <DialogContent>
   637→        <DialogHeader>
   638→          <DialogTitle>Edit Laporan</DialogTitle>
   639→          <DialogDescription>
   640→            Anda dapat mengedit judul dan deskripsi selama laporan masih berstatus PENDING.
   641→          </DialogDescription>
   642→        </DialogHeader>
   643→        <div className="space-y-3">
   644→          <div className="space-y-2">
   645→            <Label htmlFor="edit-title">Judul</Label>
   646→            <Input id="edit-title" value={title} onChange={(e) => setTitle(e.target.value)} />
   647→          </div>
   648→          <div className="space-y-2">
   649→            <Label htmlFor="edit-desc">Deskripsi</Label>
   650→            <Textarea
   651→              id="edit-desc"
   652→              rows={5}
   653→              value={description}
   654→              onChange={(e) => setDescription(e.target.value)}
   655→            />
   656→          </div>
   657→        </div>
   658→        <DialogFooter>
   659→          <Button variant="outline" onClick={() => setOpen(false)}>
   660→            Batal
   661→          </Button>
   662→          <Button onClick={handleSave} disabled={!changed || submitting}>
   663→            {submitting && <Loader2 className="size-4 animate-spin" />}
   664→            Simpan
   665→          </Button>
   666→        </DialogFooter>
   667→      </DialogContent>
   668→    </Dialog>
   669→  )
   670→}

   672→function CommentComposer({
   673→  reportId,
   674→  onCommented,
   675→}: {
   676→  reportId: string
   677→  onCommented: () => void
   678→}) {
   679→  const [message, setMessage] = React.useState('')
   680→  const [submitting, setSubmitting] = React.useState(false)
   681→
   682→  async function handleSend() {
   683→    const text = message.trim()
   684→    if (!text) return
   685→    setSubmitting(true)
   686→    try {
   687→      await apiFetch(`/api/reports/${reportId}/comments`, {
   688→        method: 'POST',
   689→        body: JSON.stringify({ message: text }),
   690→      })
   691→      setMessage('')
   692→      toast.success('Komentar ditambahkan')
   693→      onCommented()
   694→    } catch (e) {
   695→      const msg = e instanceof ApiError ? e.message : 'Gagal menambah komentar'
   696→      toast.error('Gagal menambah komentar', { description: msg })
   697→    } finally {
   698→      setSubmitting(false)
   699→    }
   700→  }
   701→
   702→  return (
   703→    <Card>
   704→      <CardHeader className="pb-3">
   705→        <CardTitle className="flex items-center gap-2 text-base">
   706→          <MessageSquare className="size-4" />
   707→          Tambah Komentar
   708→        </CardTitle>
   709→      </CardHeader>
   710→      <CardContent className="space-y-3 px-4 sm:px-6">
   711→        <Textarea
   712→          value={message}
   713→          onChange={(e) => setMessage(e.target.value)}
   714→          placeholder="Tulis komentar atau update singkat untuk laporan ini…"
   715→          rows={3}
   716→          onKeyDown={(e) => {
   717→            if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   718→              handleSend()
   719→            }
   720→          }}
   721→        />
   722→        <div className="flex items-center justify-between gap-2">
   723→          <span className="text-muted-foreground text-[11px]">
   724→            Tekan Cmd/Ctrl + Enter untuk kirim cepat
   725→          </span>
   726→          <Button onClick={handleSend} disabled={!message.trim() || submitting}>
   727→            {submitting ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   728→            Kirim Komentar
   729→          </Button>
   730→        </div>
   731→      </CardContent>
   732→    </Card>
   733→  )
   734→}

   736→function DeleteReportButton({
