   462→   462→  const [submitting, setSubmitting] = React.useState(false)
   463→   463→
   464→   464→  // Keep local state synced when report changes
   465→   465→  React.useEffect(() => {
   466→   466→    setStatus(report.status)
   467→   467→    setPriority(report.priority)
   468→   468→    setResolution(report.resolution ?? '')

   486→   486→    enabled: canAssign,
   487→   487→  })
   488→   488→
   489→   489→  const statusChanged = status !== report.status
   490→   490→  const priorityChanged = priority !== report.priority
   491→   491→  const resolutionChanged = resolution !== (report.resolution ?? '')
   492→   492→  const assigneeChanged =
   493→   493→    (assigneeId === '__none__' ? null : assigneeId) !==
   494→   494→    (report.assigneeId ?? null)
   495→   495→
   496→   496→  const hasChange =
   497→   497→    statusChanged || priorityChanged || resolutionChanged || assigneeChanged
   498→   498→
   499→   499→  async function handleSave() {

   695→   695→function CommentComposer({
   696→   696→  reportId,
   697→   697→  onCommented,
   698→   698→}: {
   699→   699→  reportId: string
   700→   700→  onCommented: () => void
   701→   701→}) {
   702→   702→  const [message, setMessage] = React.useState('')
   703→   703→  const [submitting, setSubmitting] = React.useState(false)
   704→   704→
   705→   705→  async function handleSend() {
   706→   706→    const text = message.trim()
   707→   707→    if (!text) return
   708→   708→    setSubmitting(true)
   709→   709→    try {
   710→   710→      await apiFetch(`/api/reports/${reportId}/comments`, {
   711→   711→        method: 'POST',
   712→   712→        body: JSON.stringify({ message: text }),
   713→   713→      })
   714→   714→      setMessage('')
   715→   715→      toast.success('Komentar ditambahkan')
   716→   716→      onCommented()
   717→   717→    } catch (e) {
   718→   718→      const msg = e instanceof ApiError ? e.message : 'Gagal menambah komentar'
   719→   719→      toast.error('Gagal menambah komentar', { description: msg })
   720→   720→    } finally {
   721→   721→      setSubmitting(false)
   722→   722→    }
   723→   723→  }
   724→   724→
   725→   725→  return (
   726→   726→    <Card>
   727→   727→      <CardHeader className="pb-3">
   728→   728→        <CardTitle className="flex items-center gap-2 text-base">
   729→   729→          <MessageSquare className="size-4" />
   730→   730→          Tambah Komentar
   731→   731→        </CardTitle>
   732→   732→      </CardHeader>
   733→   733→      <CardContent className="space-y-3 px-4 sm:px-6">
   734→   734→        <Textarea
   735→   735→          value={message}
   736→   736→          onChange={(e) => setMessage(e.target.value)}
   737→   737→          placeholder="Tulis komentar atau update singkat untuk laporan ini…"
   738→   738→          rows={3}
   739→   739→          onKeyDown={(e) => {
   740→   740→            if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) {
   741→   741→              handleSend()
   742→   742→            }
   743→   743→          }}
   744→   744→        />
   745→   745→        <div className="flex items-center justify-between gap-2">
   746→   746→          <span className="text-muted-foreground text-[11px]">
   747→   747→            Tekan Cmd/Ctrl + Enter untuk kirim cepat
   748→   748→          </span>
   749→   749→          <Button onClick={handleSend} disabled={!message.trim() || submitting}>
   750→   750→            {submitting ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   751→   751→            Kirim Komentar
   752→   752→          </Button>
   753→   753→        </div>
   754→   754→      </CardContent>
   755→   755→    </Card>
   756→   756→  )
